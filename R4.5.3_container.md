# build R container in HPC

- rebuilt on 31 May 2026
- based on a previous functional container https://github.com/GITC2025/integration_benchmark/blob/main/custom_sctk_container.md
- we update it with a comprehensive toolkit
- singlecelltk is not 100% functional but we use it as a way to download all related dependencies
- the patched singlecelltk script can be used for a quick automated QC metric check as needed
- HPC lockdown mode is necessary to prevent internet connection errors during usage and prevent version drift, as R has a tendency to download things on the fly
- run this on a cluster w internet access, e.g. Fir or Frontenac CAC 
- during analysis, run with external libraries to avoid remaking the container
- when analysis is complete, bundle all external libraries into the container as a mega container

# iterative compile loop to isolate and overcome failure states
install_robustly <- function(package_list) {
    for (pkg in package_list) {
        clean_name <- gsub(".*::", "", gsub(".*/", "", pkg))
        if (!requireNamespace(clean_name, quietly = TRUE)) {
            message("Compiling target dependency: ", pkg)
            tryCatch({
                pak::pkg_install(pkg, ask = FALSE)
            }, error = function(e) {
                message("ERROR: Critical failure compiling ", pkg, ": ", e$message)
            })
        } else {
            message("Targeted dependency already resolved: ", clean_name)
        }
    }
}

message("Standard Repository Resolution")
install_robustly(cran_bioc_pkgs)

message("Remote GitHub Resolution")
install_robustly(github_pkgs)

```sh
#!/bin/bash
#SBATCH --job-name=build_R_31May2026
#SBATCH --time=02:00:00
#SBATCH --cpus-per-task=8
#SBATCH --mem=128G
#SBATCH --mail-type=END,FAIL
#SBATCH --mail-user=delphine.girona@gmail.com
#SBATCH --output=/scratch/delphine/sctk_build_%j.out
#SBATCH --error=/scratch/delphine/sctk_build_%j.err

set -euo pipefail

ulimit -n $(ulimit -Hn)
ulimit -u $(ulimit -Hu)

# specify apptainer version if you wish
module load apptainer

USER_SCRATCH="/scratch/${USER}"

export APPTAINER_TMPDIR="$SLURM_TMPDIR"
export APPTAINER_CACHEDIR="${USER_SCRATCH}/apptainer_cache"

mkdir -p "$APPTAINER_CACHEDIR"
cd "$USER_SCRATCH"

apptainer build --fakeroot R_31May2026.sif R_31May2026.def

echo Build complete
```
